// ═══ src/utils/asyncHandler.js ═══
// Wraps async route handlers so thrown errors reach the error middleware
// instead of being duplicated in try/catch blocks in every route.
const asyncHandler = (fn) => (req, res, next) => {
    Promise.resolve(fn(req, res, next)).catch(next);
};

module.exports = asyncHandler;

// ═══ src/utils/httpError.js ═══
// Error type carrying an HTTP status code, used with the error middleware.
class HttpError extends Error {
    constructor(status, message) {
        super(message);
        this.status = status;
    }
}

module.exports = HttpError;

// ═══ src/middleware/errorMiddleware.js ═══
// Central error handler: replaces the per-route
// `res.status(...).json({ error: ... })` catch blocks.
const errorMiddleware = (err, req, res, next) => {
    const status = err.status || 500;
    res.status(status).json({ error: err.message || 'Internal server error' });
};

module.exports = errorMiddleware;

const SALT_ROUNDS = 10;

const hashPassword = (password) => bcrypt.hash(password, SALT_ROUNDS);

const verifyPassword = (password, passwordHash) => bcrypt.compare(password, passwordHash);

module.exports = { hashPassword, verifyPassword };

// ═══ src/utils/token.js ═══
const jwt = require('jsonwebtoken');

const JWT_SECRET = process.env.JWT_SECRET || 'change-me';
const TOKEN_TTL = '1h';

const signToken = (payload) => jwt.sign(payload, JWT_SECRET, { expiresIn: TOKEN_TTL });

const verifyToken = (token) => jwt.verify(token, JWT_SECRET);

module.exports = { signToken, verifyToken };

// ═══ src/models/schemaUtils.js ═══
const mongoose = require('mongoose');

// Base fields shared by every model.
const baseFields = {
    id: { type: String, required: true, unique: true },
    createdAt: { type: Date, default: Date.now }
};

// Build a schema with the shared base fields. Pass `withUpdatedAt: true`
// to add an updatedAt field kept current by a pre-save hook.
const createSchema = (fields, { withUpdatedAt = false } = {}) => {
    const definition = { ...baseFields, ...fields };
    if (withUpdatedAt) {
        definition.updatedAt = { type: Date, default: Date.now };
    }
    const schema = new mongoose.Schema(definition);
    if (withUpdatedAt) {
        schema.pre('save', function (next) {
            this.updatedAt = Date.now();
            next();
        });
    }
    return schema;
};

module.exports = { createSchema };

// ═══ src/middleware/authMiddleware.js ═══
const { verifyToken } = require('../utils/token');

const auth = (req, res, next) => {
    const token = req.headers['authorization']?.split(' ')[1];
    if (!token) return res.status(401).json({ error: 'Access denied' });
    try {
        req.user = verifyToken(token); // Attach user data to request
        next();
    } catch (error) {
        res.status(400).json({ error: 'Invalid token' });
    }
};

module.exports = auth;

// ═══ src/controllers/userController.js ═══
const User = require('../models/User');
const { hashPassword, verifyPassword } = require('../utils/password');
const { signToken } = require('../utils/token');
const { generateId } = require('../utils/id');
const HttpError = require('../utils/httpError');

exports.register = async (req, res) => {
    const { username, email, password } = req.body;
    const passwordHash = await hashPassword(password);
    const newUser = await User.create({ id: generateId(), username, email, passwordHash });
    res.status(201).json({ message: 'User registered successfully', userId: newUser.id });
};

exports.login = async (req, res) => {
    const { email, password } = req.body;
    const user = await User.findOne({ email });
    if (!user) throw new HttpError(404, 'User not found');
    const isValid = await verifyPassword(password, user.passwordHash);
    if (!isValid) throw new HttpError(401, 'Invalid credentials');
    const token = signToken({ id: user.id });
    res.json({ token });
};

// ═══ src/socket/chatSocket.js ═══
const socketIo = require('socket.io');
const Chat = require('../models/Chat');
const { generateId } = require('../utils/id');

module.exports = (server) => {
    const io = socketIo(server);

    io.on('connection', (socket) => {
        console.log('New client connected');
        const gameId = socket.handshake.query.gameId;

        // Join the room for the specific game
        socket.join(gameId);

        // Listen for incoming messages
        socket.on('sendMessage', async ({ userId, message }) => {
            const chatMessage = await Chat.create({ id: generateId(), gameId, userId, message });
            io.to(gameId).emit('newMessage', chatMessage);
        });

        socket.on('disconnect', () => {
            console.log('Client disconnected');
        });
    });

    return io;
};

// ═══ src/hooks/useFetch.js ═══
import { useEffect, useState } from 'react';
import client from '../api/client';

// Shared data-fetching hook: replaces the fetch-in-useEffect
// boilerplate repeated across list/detail components.
const useFetch = (url) => {
    const [data, setData] = useState(null);
    const [loading, setLoading] = useState(true);
    const [error, setError] = useState(null);

    useEffect(() => {
        let cancelled = false;
        const fetchData = async () => {
            setLoading(true);
            try {
                const response = await client.get(url);
                if (!cancelled) setData(response.data);
            } catch (e) {
                if (!cancelled) setError('Error fetching data');
            } finally {
                if (!cancelled) setLoading(false);
            }
        };
        fetchData();
        return () => { cancelled = true; };
    }, [url]);

    return { data, loading, error };
};

export default useFetch;

// ═══ src/components/GameChat.js ═══
import React, { useEffect, useRef, useState } from 'react';
import { io } from 'socket.io-client';

const GameChat = ({ gameId, userId }) => {
    const [messages, setMessages] = useState([]);
    const [message, setMessage] = useState('');
    const socketRef = useRef(null);

    useEffect(() => {
        const socket = io(process.env.REACT_APP_API_URL, { query: { gameId } });
        socketRef.current = socket;

        socket.on('newMessage', (newMessage) => {
            setMessages((prevMessages) => [...prevMessages, newMessage]);
        });

        // Fetch chat history on mount
        const fetchChatHistory = async () => {
            const response = await fetch(`/api/games/${gameId}/chat`);
            const chatHistory = await response.json();
            setMessages(chatHistory);
        };
        fetchChatHistory();

        return () => { socket.disconnect(); };
    }, [gameId]);

    const sendChatMessage = (e) => {
        e.preventDefault();
        socketRef.current.emit('sendMessage', { userId, message });
        setMessage('');
    };

    return (
        <div className="p-4 border rounded-md bg-gray-50">
            <div className="mb-2 h-48 overflow-y-auto">
                {messages.map((msg, index) => (
                    <div key={index}>[{new Date(msg.createdAt).toLocaleTimeString()}] {msg.userId}: {msg.message}</div>
                ))}
            </div>
            <form onSubmit={sendChatMessage} className="flex">
                <input
                    type="text"
                    value={message}
                    onChange={(e) => setMessage(e.target.value)}
                    className="flex-grow mr-2 p-2 border border-gray-300 rounded-md"
                    placeholder="Type your message..."
                />
                <button type="submit" className="px-4 bg-blue-500 text-white rounded-md">Send</button>
            </form>
        </div>
    );
};

export default GameChat;
